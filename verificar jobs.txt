# 1) backup do banco
cp /opt/phylogentree/shared/data/jobs.sqlite3 /opt/phylogentree/shared/data/jobs.sqlite3.bak.$(date +%F-%H%M%S)

# 2) ver quais serão cancelados
sqlite3 -header -column /opt/phylogentree/shared/data/jobs.sqlite3 \
"select id, job_name, status, created_at from jobs where status='queued' order by created_at;"

# 3) cancelar + limpar pastas
for id in $(sqlite3 /opt/phylogentree/shared/data/jobs.sqlite3 "select id from jobs where status='queued';"); do
  echo "Admin cancel: $id"
  rm -rf "/opt/phylogentree/shared/uploads/$id" "/opt/phylogentree/shared/results/$id"
  sqlite3 /opt/phylogentree/shared/data/jobs.sqlite3 \
  "update jobs
   set status='failed',
       step='failed',
       progress=0,
       error_message='admin canceled',
       completed_at=coalesce(completed_at, datetime('now')),
       expires_at=coalesce(expires_at, datetime('now', '+3 days')),
       updated_at=datetime('now')
   where id='$id';"
done

# 4) conferir resultado
sqlite3 -header -column /opt/phylogentree/shared/data/jobs.sqlite3 \
"select status, count(*) total from jobs group by status order by total desc;"
